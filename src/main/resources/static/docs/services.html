UserService

Реализует бизнес-логику работы с пользователями:

  • findAll() — получить список всех пользователей
  • findById(Long id) — найти пользователя по ID
  • save(User user) — создать или обновить пользователя
  • delete(Long id) — удалить пользователя по ID
@Service
public class UserService {

  private final UserRepository repository;

  public UserService(UserRepository repository) {
    this.repository = repository;
  }

  public List[User] findAll() {
    return repository.findAll();
  }

  public User findById(Long id) {
    return repository.findById(id).orElse(null);
  }

  public void save(User user) {
    repository.save(user);
  }

  public void delete(Long id) {
    repository.deleteById(id);
  }
}
